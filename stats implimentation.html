import numpy as np

def outlier_detection(data, threshold=2):
    mean_data = np.mean(data)
    std = np.std(data)

    outliers = [
        x for x in data
        if abs(x - mean_data) / std > threshold
    ]

    return outliers


# Example
data = [1, 2, 3, 4, 5, 7, 834, 56, 78]

print("Outliers:", outlier_detection(data))

Outliers: [834]

def q1():
    def detectoutliers_iqr(data):
        q1 = np.percentile(data, 25)
        q3 = np.percentile(data, 75)

        iqr = q3 - q1

        lower_bound = q1 - 1.5 * iqr
        upper_bound = q3 + 1.5 * iqr

        outliers = []

        for x in data:
            if x < lower_bound or x > upper_bound:
                outliers.append(x)

        return outliers

    data = [1, 2, 3, 5, 8, 9, 45, 89, 43, 67]
